    public static class CapturadorDatos
    {
        public static string LeerTextoOBLIGATORIO(string mensaje)
        {
            string entrada = "";
            while (string.IsNullOrWhiteSpace(entrada))
            {
                Console.Write($"{mensaje}: ");
                entrada = Console.ReadLine();
                if (string.IsNullOrWhiteSpace(entrada))
                {
                    Console.WriteLine("   [ERROR]: Este campo es obligatorio.");
                }
            }
            return entrada;
        }

        public static int LeerEntero(string mensaje, int minimo = 0, int maximo = 150)
        {
            int valor = 0;
            bool valido = false;
            while (!valido)
            {
                Console.Write($"{mensaje}: ");
                try
                {
                    valor = Convert.ToInt32(Console.ReadLine());
                    if (valor < minimo || valor > maximo)
                    {
                        throw new ArgumentOutOfRangeException($"El entero debe estar entre {minimo} y {maximo}.");
                    }
                    valido = true;
                }
                catch (FormatException)
                {
                    Console.WriteLine("   [ERROR FORMATO]: Ingrese un valor entero.");
                }
                catch (ArgumentOutOfRangeException ex)
                {
                    Console.WriteLine($"   [ERROR RANGO]: {ex.Message}");
                }
                finally
                {
                    if (!valido) Console.WriteLine("   --> Reintente.\n");
                }
            }
            return valor;
        }

        public static double LeerDecimal(string mensaje, double minimo = 0.0, double maximo = 500.0)
        {
            double valor = 0.0;
            bool valido = false;
            while (!valido)
            {
                Console.Write($"{mensaje}: ");
                try
                {
                    valor = Convert.ToDouble(Console.ReadLine());
                    if (valor < minimo || valor > maximo)
                    {
                        throw new ArgumentOutOfRangeException($"El valor debe estar entre {minimo} y {maximo}.");
                    }
                    valido = true;
                }
                catch (FormatException)
                {
                    Console.WriteLine("   [ERROR FORMATO]: Ingrese un número decimal (ej. 36.5).");
                }
                catch (ArgumentOutOfRangeException ex)
                {
                    Console.WriteLine($"   [ERROR RANGO]: {ex.Message}");
                }
                finally
                {
                    if (!valido) Console.WriteLine("   --> Reintente.\n");
                }
            }
            return valor;
        }

        public static bool LeerBooleano(string mensaje)
        {
            while (true)
            {
                Console.Write($"{mensaje} (S/N): ");
                string resp = Console.ReadLine().Trim().ToUpper();
                if (resp == "S") return true;
                if (resp == "N") return false;
                Console.WriteLine("   [ERROR]: Responda exclusivamente 'S' o 'N'.");
            }
        }

        public static DateTime LeerFecha(string mensaje)
        {
            DateTime fecha = DateTime.MinValue;
            bool valido = false;
            while (!valido)
            {
                Console.Write($"{mensaje} (dd/mm/yyyy): ");
                try
                {
                    fecha = DateTime.ParseExact(Console.ReadLine(), "dd/MM/yyyy", null);
                    valido = true;
                }
                catch (FormatException)
                {
                    Console.WriteLine("   [ERROR FORMATO]: Respete el formato dd/mm/yyyy.");
                }
            }
            return fecha;
        }
    }

        public void CargarUsuarios()
        {
            ConexionBD conexionBD = new ConexionBD();
            using (SqlConnection conexion = conexionBD.ObtenerConexion())
            {
                try
                {
                    string query = @"
                        SELECT 
                            idUsuario AS [ID],
                            NombreUs AS [Usuario],
                            contra AS [Contraseña],
                            Rol AS [Rol],
                            CASE WHEN Activo = 1 THEN 'Activo' ELSE 'Inactivo' END AS [Estado]
                        FROM Usuarios";

                    SqlDataAdapter adapter = new SqlDataAdapter(query, conexion);
                    DataTable dt = new DataTable();
                    adapter.Fill(dt);

                    dgvUsuarios.DataSource = dt;
                }
                catch (Exception ex)
                {
                    MessageBox.Show($"Error al cargar usuarios: {ex.Message}", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
        }

                return;

            int idUsuario = Convert.ToInt32(dgvUsuarios.SelectedRows[0].Cells["ID"].Value);
            string usuario = dgvUsuarios.SelectedRows[0].Cells["Usuario"].Value.ToString();
            string contra = dgvUsuarios.SelectedRows[0].Cells["Contraseña"].Value.ToString();
            string rol = dgvUsuarios.SelectedRows[0].Cells["Rol"].Value.ToString();

            editUsers frm = new editUsers(this, idUsuario, usuario, contra, rol);
            frm.ShowDialog();
        }

        private void BtnDesactivar_Click(object sender, EventArgs e)
        {
            if (dgvUsuarios.SelectedRows.Count == 0)
            {
                MessageBox.Show("Seleccione un usuario de la lista para cambiar su estado.", "Advertencia", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            int idUsuario = Convert.ToInt32(dgvUsuarios.SelectedRows[0].Cells["ID"].Value);
            string usuario = dgvUsuarios.SelectedRows[0].Cells["Usuario"].Value.ToString();
            string estadoActual = dgvUsuarios.SelectedRows[0].Cells["Estado"].Value.ToString();

            bool nuevoEstado = estadoActual != "Activo";
            string accionText = nuevoEstado ? "activar" : "desactivar";

            DialogResult confirmacion = MessageBox.Show(
                $"¿Está seguro de que desea {accionText} al usuario '{usuario}'?",
                "Confirmación",
                MessageBoxButtons.YesNo,
                MessageBoxIcon.Question
            );

            if (confirmacion == DialogResult.Yes)
            {
                ConexionBD conexionBD = new ConexionBD();
                using (SqlConnection conexion = conexionBD.ObtenerConexion())
                {
                    try
                    {
                        conexion.Open();
                        string query = "UPDATE Usuarios SET Activo = @Activo WHERE idUsuario = @idUsuario";
                        using (SqlCommand cmd = new SqlCommand(query, conexion))
                        {
                            cmd.Parameters.AddWithValue("@Activo", nuevoEstado ? 1 : 0);
                            cmd.Parameters.AddWithValue("@idUsuario", idUsuario);
                            cmd.ExecuteNonQuery();
                        }

                        MessageBox.Show($"Usuario '{usuario}' actualizado a {(nuevoEstado ? "Activo" : "Inactivo")}.", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                        CargarUsuarios();
                    }
                    catch (Exception ex)
                    {
                        MessageBox.Show($"Error al cambiar el estado del usuario: {ex.Message}", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    }
                }
            }
        }

namespace Prueba_Proyecto_3
{
    public class GestorExpedientes : IRepositorioPaciente
    {

        // 1. Instancia estática privada única
        private static GestorExpedientes _instancia;
        private static readonly object _lock = new object();

        // 2. Lista privada de pacientes
        private List<Paciente> listaPacientes = new List<Paciente>();

        // 3. Constructor privado para evitar instanciación directa con 'new'
        private GestorExpedientes() { }

        // 4. Método público estático para obtener la única instancia 
        public static GestorExpedientes Instancia
        {
            get
            {
                lock (_lock)
                {
                    if (_instancia == null)
                    {
                        _instancia = new GestorExpedientes();
                    }
                    return _instancia;
                }
            }
        }

        public void RegistrarPaciente(Paciente nuevoPaciente)
        {
            listaPacientes.Add(nuevoPaciente);
            Console.WriteLine($"\n[ÉXITO]: Paciente {nuevoPaciente.Nombre} guardado con Expediente N° {nuevoPaciente.NumeroExpediente}.");
        }

        public void MostrarTodos()
        {
            if (listaPacientes.Count == 0)
            {
                Console.WriteLine("\nNo hay expedientes en la lista.");
                return;
            }

            foreach (var p in listaPacientes)
            {
                p.MostrarHistorialCompleto();
            }
        }

        public Paciente BuscarPorExpediente(string numExpediente)
        {
            return listaPacientes.Find(p => p.NumeroExpediente.Equals(numExpediente, StringComparison.OrdinalIgnoreCase));
        }

        public void LimpiarLista()
        {
            listaPacientes.Clear();
        }
    }
}

        private void BtnIngresar_Click(object sender, EventArgs e)
        {
            string usuario = txtUsuario.Text.Trim();
            string password = txtContra.Text.Trim();

            if (string.IsNullOrEmpty(usuario) || string.IsNullOrEmpty(password))
            {
                MessageBox.Show("Por favor, ingrese el usuario y la contraseña.", "Advertencia", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            ConexionBD conexionBD = new ConexionBD();
            using (SqlConnection conexion = conexionBD.ObtenerConexion())
            {
                try
                {
                    conexion.Open();
                    string query = "SELECT Rol FROM Usuarios WHERE NombreUs = @usuario AND contra = @contra";
                    using (SqlCommand cmd = new SqlCommand(query, conexion))
                    {
                        cmd.Parameters.AddWithValue("@usuario", usuario);
                        cmd.Parameters.AddWithValue("@contra", password);

                        object result = cmd.ExecuteScalar();

                        if (result != null)
                        {
                            string rolObtenido = result.ToString();

                            // Guardar datos de sesión
                            SesionUsuario.Instancia.IniciarSesion(usuario, rolObtenido);

                            MessageBox.Show($"Bienvenido/a, {usuario} [{rolObtenido}]", "Sesión Exitosa", MessageBoxButtons.OK, MessageBoxIcon.Information);

                            // Marcamos OK y cerramos este diálogo
                            this.DialogResult = DialogResult.OK;
                            this.Close();
                        }
                        else
                        {
                            MessageBox.Show("Usuario o contraseña incorrectos.", "Error de Autenticación", MessageBoxButtons.OK, MessageBoxIcon.Error);
                        }
                    }
                }
                catch (Exception ex)
                {
                    MessageBox.Show($"Error al conectar con la base de datos: {ex.Message}", "Error de Sistema", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
        }
    }

namespace Prueba_Proyecto_3
{

    // Estructura para registrar si cada área examinada es normal o anormal
    public class HallazgoExamen
    {
        public bool EsNormal { get; set; } = true;
        public string Detalle { get; set; } = "Sin hallazgos";
    }

    // Estructura para medicamentos recetados
    public class Medicamento
    {
        [Key]
        public int Id { get; set; }
        public string Nombre { get; set; }
        public string CantidadYDosis { get; set; }

        public Medicamento(string nombre, string cantidadYDosis)
        {
            Nombre = nombre;
            CantidadYDosis = cantidadYDosis;
        }
    }

    // Estructura para antecedentes personales patológicos. 
    public class AntecedentePersonal
    {

        [Key]
        public int Id { get; set; }
        public string Enfermedad { get; set; }
        public int AnioInicio { get; set; }
        public string TipoMedication { get; set; }
    }

    // Clase base abstracta Paciente
    public abstract class Paciente: IPaciente
    {
        [Key]
        public int Id { get; set; }

        // Datos generales
        public string Nombre { get; set; }
        public int Edad { get; set; }
        public string NumeroExpediente { get; set; }
        public string Sexo { get; set; }
        public string Direccion { get; set; }
        public string Responsable { get; set; }
        public DateTime FechaConsulta { get; set; } = DateTime.Now;

        // Motivo de Consulta
        public string ConsultaPor { get; set; }
        public string PresenteEnfermedad { get; set; }

        // Signos Vitales y Medidas Antropométricas
        public double TemperaturaCelsius { get; set; }
        public int FrecuenciaRespiratoria { get; set; }
        public int FrecuenciaCardiaca { get; set; }
        public int Pulso { get; set; }
        public string PresionArterial { get; set; }
        public double SaturacionOxigeno { get; set; }
        public double PesoKg { get; set; }
        public double TallaCm { get; set; }
        public double CircunferenciaAbdominalCm { get; set; }

        // Cálculo de IMC
        public double IMC
        {
            get
            {
                if (TallaCm <= 0) return 0;
                double tallaMetros = TallaCm / 100.0;
                return Math.Round(PesoKg / (tallaMetros * tallaMetros), 2);
            }
        }

        // Examen Físico (Opciones entre Normal/Anormal)
        public Dictionary<string, HallazgoExamen> ExamenFisico { get; set; } = new Dictionary<string, HallazgoExamen>();

        // Diagnóstico y Tratamiento
        public string TipoDiagnostico { get; set; } // Principal o Secundario
        public string DiagnosticoTexto { get; set; }
        public string TipoConsulta { get; set; } // Primera vez o Subsecuente
        public List<Medicamento> Receta { get; set; } = new List<Medicamento>();
        public string Recomendaciones { get; set; }

        public Paciente()
        {

        }
        // Constructor base
        public Paciente(string nombre, int edad, string expediente, string sexo, string direccion, string responsable)
        {
            Nombre = nombre;
            Edad = edad;
            NumeroExpediente = expediente;
            Sexo = sexo;
            Direccion = direccion;
            Responsable = responsable;
        }
        

        public virtual void MostrarHistorialCompleto()
        {
            Console.WriteLine("\n=======================================================");
            Console.WriteLine($"      EXPEDIENTE MÉDICO: {NumeroExpediente}");
            Console.WriteLine("=======================================================");
            Console.WriteLine($"Fecha Consulta: {FechaConsulta.ToShortDateString()}");
            Console.WriteLine($"Paciente: {Nombre} | Edad: {Edad} años | Sexo: {Sexo}");
            Console.WriteLine($"Dirección: {Direccion} | Responsable/Tutor: {Responsable}");

            Console.WriteLine("\n--- MOTIVO DE CONSULTA ---");
            Console.WriteLine($"Consulta Por: {ConsultaPor}");
            Console.WriteLine($"Presente Enfermedad: {PresenteEnfermedad}");

            Console.WriteLine("\n--- SIGNOS VITALES Y MEDIDAS ---");
            Console.WriteLine($"Temp: {TemperaturaCelsius}°C | FC: {FrecuenciaCardiaca} lpm | Pulso: {Pulso} | FR: {FrecuenciaRespiratoria} rpm");
            Console.WriteLine($"PA: {PresionArterial} mmHg | SpO2: {SaturacionOxigeno}%");
            Console.WriteLine($"Peso: {PesoKg} kg | Talla: {TallaCm} cm | IMC: {IMC} | Cir. Abdominal: {CircunferenciaAbdominalCm} cm");

            Console.WriteLine("\n--- EXAMEN FÍSICO ---");
            foreach (var area in ExamenFisico)
            {
                string estado = area.Value.EsNormal ? "[NORMAL]" : "[ANORMAL]";
                Console.WriteLine($" - {area.Key.PadRight(22)}: {estado} -> {area.Value.Detalle}");
            }

            Console.WriteLine("\n--- DIAGNÓSTICO Y TRATAMIENTO ---");
            Console.WriteLine($"Tipo Diagnóstico: {TipoDiagnostico} | Consulta: {TipoConsulta}");
            Console.WriteLine($"Diagnóstico: {DiagnosticoTexto}");
            Console.WriteLine("Receta Médica:");
            if (Receta.Count == 0)
            {
                Console.WriteLine("  (No se prescribieron medicamentos)");
            }
            else
            {
                for (int i = 0; i < Receta.Count; i++)
                {
                    Console.WriteLine($"  {i + 1}. {Receta[i].Nombre} - Dosis/Cant: {Receta[i].CantidadYDosis}");
                }
            }
            Console.WriteLine($"Recomendaciones: {Recomendaciones}");
        }
    
        // Destructor
        ~Paciente()
        {
            Console.WriteLine($"\n[DESTRUCTOR]: El expediente N° {NumeroExpediente} de '{Nombre}' ha sido liberado de la memoria RAM.");
        }
    }

    static class Program
    {
        [STAThread]
        static void Main()
        {
            Application.EnableVisualStyles();
            Application.SetCompatibleTextRenderingDefault(false);

            while (true)
            {
                // Reiniciamos la bandera de sesión cada vez que mostramos el Login
                LoginForm.SesionCerradaParaReinicio = false;

                using (LoginForm login = new LoginForm())
                {
                    if (login.ShowDialog() == DialogResult.OK)
                    {
                        using (MainForm principal = new MainForm())
                        {
                            principal.ShowDialog();
                        }

                        // Si al cerrar MainForm NO fue por darle a "Cerrar Sesión", salimos del bucle
                        if (!LoginForm.SesionCerradaParaReinicio)
                        {
                            break;
                        }
                    }
                    else
                    {
                        // Si se cerró la ventana de Login con la 'X' o se canceló, sale de la app
                        break;
                    }
                }
            }
        }
    }

        private bool ValidarCamposFormulario(out int edad, out double temp, out int fr, out int fc, out int pulso, out double spo2, out double peso, out double talla, out double cir)
        {
            edad = 0; temp = 0; fr = 0; fc = 0; pulso = 0; spo2 = 0; peso = 0; talla = 0; cir = 0;

            try
            {
                // 1. Validar Edad numéricamente
                if (!int.TryParse(txtEdad.Text.Trim(), out edad))
                {
                    MessageBox.Show("La edad ingresada debe ser un número entero válido.", "Error de Validación", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    txtEdad.Focus();
                    return false;
                }

                // 2. Validar Datos Generales Básicos
                ValidadorClinico.ValidarDatosGenerales(
                    txtExpediente.Text.Trim(),
                    txtNombre.Text.Trim(),
                    edad,
                    cmbSexo?.SelectedItem?.ToString() ?? "Femenino" // En embarazo/mujer siempre es femenino
                );

                // 3. Validar Motivo de Consulta
                ValidadorClinico.ValidarMotivoConsulta(
                    txtConsultaPor.Text.Trim(),
                    txtPresenteEnfermedad.Text.Trim()
                );

                // 4. Parseo de Signos Vitales
                if (!double.TryParse(txtTemp.Text.Trim(), out temp) ||
                    !int.TryParse(txtFR.Text.Trim(), out fr) ||
                    !int.TryParse(txtFC.Text.Trim(), out fc) ||
                    !int.TryParse(txtPulso.Text.Trim(), out pulso) ||
                    !double.TryParse(txtSpO2.Text.Trim(), out spo2) ||
                    !double.TryParse(txtPeso.Text.Trim(), out peso) ||
                    !double.TryParse(txtTalla.Text.Trim(), out talla) ||
                    !double.TryParse(txtCirAbdominal.Text.Trim(), out cir))
                {
                    MessageBox.Show("Uno o más campos de Signos Vitales o Medidas contienen formatos numéricos incorrectos o están vacíos.", "Error de Formato", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return false;
                }

                // 5. Validar Rangos Biológicos
                ValidadorClinico.ValidarSignosVitales(temp, fr, fc, pulso, txtPA.Text.Trim(), spo2, peso, talla, cir);

                return true;
            }
            catch (ExcepcionValidacionClinica ex)
            {
                MessageBox.Show(ex.Message, "Validación Clínica Requerida", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return false;
            }
        }
        private void BGuardar_Click(object sender, EventArgs e)
        {
            // 1. Ejecutar validación general y parseo de signos vitales
            if (!ValidarCamposFormulario(out int edad, out double temp, out int fr, out int fc, out int pulso, out double spo2, out double peso, out double talla, out double cir))
            {
                return;
            }

            // 2. Validación de rango de edad específica para Adolescente
            if (edad < 12 || edad > 17)
            {
                MessageBox.Show("Para paciente adolescente, la edad debe estar entre 12 y 17 años.", "Error de Rango", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                txtEdad.Focus();
                return;
            }

            try
            {
                // Instanciación de PacienteAdolescente usando las variables ya validadas
                PacienteAdolescente paciente = new PacienteAdolescente(
                    txtNombre.Text.Trim(),
                    edad,
                    txtExpediente.Text.Trim(),
                    cmbSexo.SelectedItem.ToString(),
                    string.IsNullOrWhiteSpace(txtDireccion.Text) ? "N/A" : txtDireccion.Text.Trim(),
                    string.IsNullOrWhiteSpace(txtResponsable.Text) ? "N/A" : txtResponsable.Text.Trim()
                );

                // Motivo y enfermedad actual
                paciente.ConsultaPor = txtConsultaPor.Text.Trim();
                paciente.PresenteEnfermedad = txtPresenteEnfermedad.Text.Trim();

                // Mapeo del Examen Físico completo
                string[] regiones = new string[]
                {
            "Apariencia general", "Cabeza", "Cuello",
            "Tórax y pulmonar", "Cardíaco", "Abdomen",
            "Columna Vertebral", "Región Glútea", "Genitales",
            "Extremidades", "Piel y anexos", "Neurologico"
                };

                foreach (var reg in regiones)
                {
                    string estadoSeleccionado = cmbsExamenFisicoEstado[reg].SelectedItem.ToString();
                    bool esNormal = (estadoSeleccionado == "Normal");
                    string detalle = txtsExamenFisicoDetalle[reg].Text.Trim();

                    if (string.IsNullOrWhiteSpace(detalle))
                    {
                        detalle = esNormal ? "Sin hallazgos / Normal" : "Alteración no especificada";
                    }

                    paciente.ExamenFisico[reg] = new HallazgoExamen
                    {
                        EsNormal = esNormal,
                        Detalle = detalle
                    };
                }

                // Asignar los signos vitales ya numéricos
                paciente.TemperaturaCelsius = temp;
                paciente.FrecuenciaRespiratoria = fr;
                paciente.FrecuenciaCardiaca = fc;
                paciente.Pulso = pulso;
                paciente.PresionArterial = string.IsNullOrWhiteSpace(txtPA.Text) ? "110/70" : txtPA.Text.Trim();
                paciente.SaturacionOxigeno = spo2;
                paciente.PesoKg = peso;
                paciente.TallaCm = talla;
                paciente.CircunferenciaAbdominalCm = cir;

                // Antecedentes Familiares Patológicos
                foreach (var kvp in chksAntFam)
                {
                    paciente.AntecedentesFamiliares[kvp.Key] = kvp.Value.Checked;
                }

                // Antecedentes Personales Patológicos
                foreach (var kvp in chksAntPers)
                {
                    if (kvp.Value.Checked)
                    {
                        int.TryParse(txtsAntPersAnio[kvp.Key].Text, out int anioInicio);
                        string medicamento = txtsAntPersMed[kvp.Key].Text.Trim();

                        paciente.AntecedentesPersonales.Add(new AntecedentePersonal
                        {
                            Enfermedad = kvp.Key,
                            AnioInicio = anioInicio,
                            TipoMedication = string.IsNullOrWhiteSpace(medicamento) ? "N/A" : medicamento
                        });
                    }
                }

                // Estilos de vida y consumo
                paciente.Etilista = chkEtilista.Checked;
                paciente.Tabaquista = chkTabaquista.Checked;
                paciente.Drogadiccion = chkDrogadiccion.Checked;

                // Evaluaciones adicionales
                paciente.EscalaTanner = string.IsNullOrWhiteSpace(txtEscalaTanner.Text) ? "N/A" : txtEscalaTanner.Text.Trim();
                paciente.PanelSnellen = string.IsNullOrWhiteSpace(txtPanelSnellen.Text) ? "N/A" : txtPanelSnellen.Text.Trim();

                // Solicitud de Laboratorios
                if (chkLabHemograma.Checked) paciente.LaboratoriosSolicitados.Add("Hemograma automatizado (Muestra: Sangre)");
                if (chkLabColesterol.Checked) paciente.LaboratoriosSolicitados.Add("Colesterol (Muestra: Sangre)");
                if (chkTrigliceridos.Checked) paciente.LaboratoriosSolicitados.Add("Triglicéridos (Muestra: Sangre)");
                if (chkLabHeces.Checked) paciente.LaboratoriosSolicitados.Add("General de Heces (Muestra: Heces)");
                if (chkLabOrina.Checked) paciente.LaboratoriosSolicitados.Add("General de Orina (Muestra: Orina)");

                // Diagnóstico, tipo de consulta y tratamiento
                paciente.DiagnosticoTexto = txtDiagnostico.Text.Trim();
                paciente.TipoDiagnostico = cmbTipoDiagnostico.SelectedItem.ToString();
                paciente.TipoConsulta = cmbTipoConsulta.SelectedItem.ToString();
                paciente.Recomendaciones = txtRecomendaciones.Text.Trim();

                if (!string.IsNullOrWhiteSpace(txtReceta.Text))
                {
                    paciente.Receta.Add(new Medicamento("Medicamento Adolescente", txtReceta.Text.Trim()));
                }

                // Guardar mediante Repositorio
                repositorio.RegistrarPaciente(paciente);

                MessageBox.Show("¡Expediente de Paciente Adolescente guardado con éxito!", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                this.Close();
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ocurrió un error al guardar el registro: {ex.Message}", "Error de Sistema", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private int ObtenerValorBarthel(ComboBox cmb)

        private bool ValidarCamposFormulario(out int edad, out double temp, out int fr, out int fc, out int pulso, out double spo2, out double peso, out double talla, out double cir)
        {
            edad = 0; temp = 0; fr = 0; fc = 0; pulso = 0; spo2 = 0; peso = 0; talla = 0; cir = 0;

            try
            {
                if (!int.TryParse(txtEdad.Text.Trim(), out edad))
                {
                    MessageBox.Show("La edad ingresada debe ser un número entero válido.", "Error de Validación", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    txtEdad.Focus();
                    return false;
                }
                ValidadorClinico.ValidarDatosGenerales(
                    txtExpediente.Text.Trim(),
                    txtNombre.Text.Trim(),
                    edad,
                    cmbSexo?.SelectedItem?.ToString() ?? "Femenino"
                );

                ValidadorClinico.ValidarMotivoConsulta(
                    txtConsultaPor.Text.Trim(),
                    txtPresenteEnfermedad.Text.Trim()
                );
                if (!double.TryParse(txtTemp.Text.Trim(), out temp) ||
                    !int.TryParse(txtFR.Text.Trim(), out fr) ||
                    !int.TryParse(txtFC.Text.Trim(), out fc) ||
                    !int.TryParse(txtPulso.Text.Trim(), out pulso) ||
                    !double.TryParse(txtSpO2.Text.Trim(), out spo2) ||
                    !double.TryParse(txtPeso.Text.Trim(), out peso) ||
                    !double.TryParse(txtTalla.Text.Trim(), out talla) ||
                    !double.TryParse(txtCirAbdominal.Text.Trim(), out cir))
                {
                    MessageBox.Show("Uno o más campos de Signos Vitales o Medidas contienen formatos numéricos incorrectos o están vacíos.", "Error de Formato", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return false;
                }

                ValidadorClinico.ValidarSignosVitales(temp, fr, fc, pulso, txtPA.Text.Trim(), spo2, peso, talla, cir);

                return true;
            }
            catch (ExcepcionValidacionClinica ex)
            {
                MessageBox.Show(ex.Message, "Validación Clínica Requerida", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return false;
            }
        }
        private void BGuardar_Click(object sender, EventArgs e)
        {
            if (!ValidarCamposFormulario(out int edad, out double temp, out int fr, out int fc, out int pulso, out double spo2, out double peso, out double talla, out double cir))
            {
                return;
            }

            if (edad < 60 || edad > 120)
            {
                MessageBox.Show("Para el paciente adulto mayor, la edad debe estar entre 60 y 120 años.", "Error de Rango", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                txtEdad.Focus();
                return;
            }

            try
            {
                PacienteAdultoMayor paciente = new PacienteAdultoMayor(
                    txtNombre.Text.Trim(),
                    edad,
                    txtExpediente.Text.Trim(),
                    cmbSexo.SelectedItem.ToString(),
                    string.IsNullOrWhiteSpace(txtDireccion.Text) ? "N/A" : txtDireccion.Text.Trim(),
                    string.IsNullOrWhiteSpace(txtResponsable.Text) ? "N/A" : txtResponsable.Text.Trim()
                );

                paciente.ConsultaPor = txtConsultaPor.Text.Trim();
                paciente.PresenteEnfermedad = txtPresenteEnfermedad.Text.Trim();
                string[] regiones = new string[]
                {
            "Apariencia general", "Cabeza", "Cuello",
            "Tórax y pulmonar", "Cardíaco", "Abdomen",
            "Columna Vertebral", "Región Glútea", "Genitales",
            "Extremidades", "Piel y anexos", "Neurologico"
                };

                foreach (var reg in regiones)
                {
                    string estadoSeleccionado = cmbsExamenFisicoEstado[reg].SelectedItem.ToString();
                    bool esNormal = (estadoSeleccionado == "Normal");
                    string detalle = txtsExamenFisicoDetalle[reg].Text.Trim();

                    if (string.IsNullOrWhiteSpace(detalle))
                    {
                        detalle = esNormal ? "Sin hallazgos / Normal" : "Alteración no especificada";
                    }

                    paciente.ExamenFisico[reg] = new HallazgoExamen
                    {
                        EsNormal = esNormal,
                        Detalle = detalle
                    };
                }

                paciente.TemperaturaCelsius = temp;
                paciente.FrecuenciaRespiratoria = fr;
                paciente.FrecuenciaCardiaca = fc;
                paciente.Pulso = pulso;
                paciente.PresionArterial = string.IsNullOrWhiteSpace(txtPA.Text) ? "130/85" : txtPA.Text.Trim();
                paciente.SaturacionOxigeno = spo2;
                paciente.PesoKg = peso;
                paciente.TallaCm = talla;
                paciente.CircunferenciaAbdominalCm = cir;

                // Cálculo automático del Índice de Barthel
                int sumaBarthel = ObtenerValorBarthel(cmbComer) +
                                  ObtenerValorBarthel(cmbBaniarse) +
                                  ObtenerValorBarthel(cmbVestirse) +
                                  ObtenerValorBarthel(cmbAseo) +
                                  ObtenerValorBarthel(cmbDeposicion) +
                                  ObtenerValorBarthel(cmbMiccion) +
                                  ObtenerValorBarthel(cmbRetrete) +
                                  ObtenerValorBarthel(cmbTrasladarse) +
                                  ObtenerValorBarthel(cmbDeambulacion) +
                                  ObtenerValorBarthel(cmbEscalones);

                paciente.PuntajeBarthel = sumaBarthel;

                if (sumaBarthel < 20) paciente.CategoriaDependencia = "Dependencia Total";
                else if (sumaBarthel <= 35) paciente.CategoriaDependencia = "Dependencia Grave";
                else if (sumaBarthel <= 55) paciente.CategoriaDependencia = "Dependencia Moderada";
                else if (sumaBarthel <= 90) paciente.CategoriaDependencia = "Dependencia Leve";
                else paciente.CategoriaDependencia = "Independiente";

                paciente.DiagnosticoTexto = txtDiagnostico.Text.Trim();
                paciente.TipoDiagnostico = cmbTipoDiagnostico.SelectedItem.ToString();
                paciente.TipoConsulta = cmbTipoConsulta.SelectedItem.ToString();
                paciente.Recomendaciones = txtRecomendaciones.Text.Trim();

                if (!string.IsNullOrWhiteSpace(txtReceta.Text))
                {
                    paciente.Receta.Add(new Medicamento("Medicamento Geriátrico", txtReceta.Text.Trim()));
                }

                repositorio.RegistrarPaciente(paciente);

                MessageBox.Show($"¡Expediente de Adulto Mayor guardado con éxito!\nPuntaje Barthel: {sumaBarthel} pts ({paciente.CategoriaDependencia})", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                this.Close();
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ocurrió un error al guardar el registro: {ex.Message}", "Error de Sistema", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private bool ValidarCamposFormulario(out int edad, out double temp, out int fr, out int fc, out int pulso, out double spo2, out double peso, out double talla, out double cir)
        {
            edad = 0; temp = 0; fr = 0; fc = 0; pulso = 0; spo2 = 0; peso = 0; talla = 0; cir = 0;

            try
            {
                if (!int.TryParse(txtEdad.Text.Trim(), out edad))
                {
                    MessageBox.Show("La edad ingresada debe ser un número entero válido.", "Error de Validación", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    txtEdad.Focus();
                    return false;
                }
                ValidadorClinico.ValidarDatosGenerales(
                    txtExpediente.Text.Trim(),
                    txtNombre.Text.Trim(),
                    edad,
                    cmbSexo?.SelectedItem?.ToString() ?? "Femenino"
                );

                ValidadorClinico.ValidarMotivoConsulta(
                    txtConsultaPor.Text.Trim(),
                    txtPresenteEnfermedad.Text.Trim()
                );
                if (!double.TryParse(txtTemp.Text.Trim(), out temp) ||
                    !int.TryParse(txtFR.Text.Trim(), out fr) ||
                    !int.TryParse(txtFC.Text.Trim(), out fc) ||
                    !int.TryParse(txtPulso.Text.Trim(), out pulso) ||
                    !double.TryParse(txtSpO2.Text.Trim(), out spo2) ||
                    !double.TryParse(txtPeso.Text.Trim(), out peso) ||
                    !double.TryParse(txtTalla.Text.Trim(), out talla) ||
                    !double.TryParse(txtCirAbdominal.Text.Trim(), out cir))
                {
                    MessageBox.Show("Uno o más campos de Signos Vitales o Medidas contienen formatos numéricos incorrectos o están vacíos.", "Error de Formato", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return false;
                }

                ValidadorClinico.ValidarSignosVitales(temp, fr, fc, pulso, txtPA.Text.Trim(), spo2, peso, talla, cir);

                return true;
            }
            catch (ExcepcionValidacionClinica ex)
            {
                MessageBox.Show(ex.Message, "Validación Clínica Requerida", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return false;
            }
        }
        private void BGuardar_Click(object sender, EventArgs e)
        {
            //Ejecutar validaciones generales y parseo de signos vitales
            if (!ValidarCamposFormulario(out int edad, out double temp, out int fr, out int fc, out int pulso, out double spo2, out double peso, out double talla, out double cir))
            {
                return;
            }

            try
            {
                //Instanciación del Paciente Crónico con los datos ya validados
                PacienteCronico paciente = new PacienteCronico(
                    txtNombre.Text.Trim(),
                    edad,
                    txtExpediente.Text.Trim(),
                    cmbSexo.SelectedItem.ToString(),
                    string.IsNullOrWhiteSpace(txtDireccion.Text) ? "N/A" : txtDireccion.Text.Trim(),
                    string.IsNullOrWhiteSpace(txtResponsable.Text) ? "N/A" : txtResponsable.Text.Trim()
                );

                //Motivo de consulta
                paciente.ConsultaPor = txtConsultaPor.Text.Trim();
                paciente.PresenteEnfermedad = txtPresenteEnfermedad.Text.Trim();

                //Mapeo del Examen Físico completo por regiones
                string[] regiones = new string[]
                {
            "Apariencia general", "Cabeza", "Cuello",
            "Tórax y pulmonar", "Cardíaco", "Abdomen",
            "Columna Vertebral", "Región Glútea", "Genitales",
            "Extremidades", "Piel y anexos", "Neurologico"
                };

                foreach (var reg in regiones)
                {
                    string estadoSeleccionado = cmbsExamenFisicoEstado[reg].SelectedItem.ToString();
                    bool esNormal = (estadoSeleccionado == "Normal");
                    string detalle = txtsExamenFisicoDetalle[reg].Text.Trim();

                    if (string.IsNullOrWhiteSpace(detalle))
                    {
                        detalle = esNormal ? "Sin hallazgos / Normal" : "Alteración no especificada";
                    }

                    paciente.ExamenFisico[reg] = new HallazgoExamen
                    {
                        EsNormal = esNormal,
                        Detalle = detalle
                    };
                }

                //Asignación de signos vitales y medidas antropométricas
                paciente.TemperaturaCelsius = temp;
                paciente.FrecuenciaRespiratoria = fr;
                paciente.FrecuenciaCardiaca = fc;
                paciente.Pulso = pulso;
                paciente.PresionArterial = string.IsNullOrWhiteSpace(txtPA.Text) ? "120/80" : txtPA.Text.Trim();
                paciente.SaturacionOxigeno = spo2;
                paciente.PesoKg = peso;
                paciente.TallaCm = talla;
                paciente.CircunferenciaAbdominalCm = cir;

                //Diagnóstico, tipo de consulta y prescripción
                paciente.DiagnosticoTexto = txtDiagnostico.Text.Trim();
                paciente.TipoDiagnostico = cmbTipoDiagnostico.SelectedItem.ToString();
                paciente.TipoConsulta = cmbTipoConsulta.SelectedItem.ToString();
                paciente.Recomendaciones = txtRecomendaciones.Text.Trim();

                if (!string.IsNullOrWhiteSpace(txtReceta.Text))
                {
                    paciente.Receta.Add(new Medicamento("Medicamento Crónico", txtReceta.Text.Trim()));
                }

                //Persistencia a través del repositorio
                repositorio.RegistrarPaciente(paciente);

                MessageBox.Show("¡Expediente de Paciente Crónico guardado con éxito en la base de datos!", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                this.Close();
            }
            catch (Exception ex)
            {
                string errorReal = ex.InnerException != null ? ex.InnerException.Message : ex.Message;
                if (ex.InnerException != null && ex.InnerException.InnerException != null)
                {
                    errorReal += " -> " + ex.InnerException.InnerException.Message;
                }

                MessageBox.Show($"Detalle del error: {errorReal}", "Error de Sistema", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }
    }

        private void BGuardar_Click(object sender, EventArgs e)
        {
            try
            {
                if (string.IsNullOrWhiteSpace(txtNombre.Text) || string.IsNullOrWhiteSpace(txtExpediente.Text))
                {
                    MessageBox.Show("El nombre y el número de expediente son obligatorios.", "Advertencia", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return;
                }

                if (!int.TryParse(txtEdad.Text, out int edad) || edad < 10 || edad > 55)
                {
                    MessageBox.Show("Para paciente embarazada, la edad debe estar entre 10 y 55 años.", "Error de Rango", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    return;
                }

                // Instanciación de PacienteEmbarazada
                PacienteEmbarazada paciente = new PacienteEmbarazada(
                    txtNombre.Text.Trim(),
                    edad,
                    txtExpediente.Text.Trim(),
                    string.IsNullOrWhiteSpace(txtDireccion.Text) ? "N/A" : txtDireccion.Text.Trim(),
                    string.IsNullOrWhiteSpace(txtResponsable.Text) ? "N/A" : txtResponsable.Text.Trim()
                );

                // Motivo
                paciente.ConsultaPor = txtConsultaPor.Text.Trim();
                paciente.PresenteEnfermedad = txtPresenteEnfermedad.Text.Trim();

                // Mapeo del Examen Físico completo
                string[] regiones = new string[]
                {
                    "Apariencia general", "Cabeza", "Cuello",
                    "Tórax y pulmonar", "Cardíaco", "Abdomen",
                    "Columna Vertebral", "Región Glútea", "Genitales",
                    "Extremidades", "Piel y anexos", "Neurologico"
                };

                foreach (var reg in regiones)
                {
                    string estadoSeleccionado = cmbsExamenFisicoEstado[reg].SelectedItem.ToString();
                    bool esNormal = (estadoSeleccionado == "Normal");
                    string detalle = txtsExamenFisicoDetalle[reg].Text.Trim();

                    if (string.IsNullOrWhiteSpace(detalle))
                    {
                        detalle = esNormal ? "Sin hallazgos / Normal" : "Alteración no especificada";
                    }

                    paciente.ExamenFisico[reg] = new HallazgoExamen
                    {
                        EsNormal = esNormal,
                        Detalle = detalle
                    };
                }

                // Signos vitales
                double.TryParse(txtTemp.Text, out double temp);
                int.TryParse(txtFR.Text, out int fr);
                int.TryParse(txtFC.Text, out int fc);
                int.TryParse(txtPulso.Text, out int pulso);
                double.TryParse(txtSpO2.Text, out double spo2);
                double.TryParse(txtPeso.Text, out double peso);
                double.TryParse(txtTalla.Text, out double talla);
                double.TryParse(txtCirAbdominal.Text, out double cir);
                double.TryParse(txtPesoAnterior.Text, out double pesoAnt);

                paciente.TemperaturaCelsius = temp;
                paciente.FrecuenciaRespiratoria = fr;
                paciente.FrecuenciaCardiaca = fc;
                paciente.Pulso = pulso;
                paciente.PresionArterial = string.IsNullOrWhiteSpace(txtPA.Text) ? "120/80" : txtPA.Text.Trim();
                paciente.SaturacionOxigeno = spo2;
                paciente.PesoKg = peso;
                paciente.TallaCm = talla;
                paciente.CircunferenciaAbdominalCm = cir;

                // Datos obstétricos
                paciente.FormulaObstetrica = string.IsNullOrWhiteSpace(txtFormulaObstetrica.Text) ? "G1 P0 A0 V0" : txtFormulaObstetrica.Text.Trim();
                paciente.FinEmbarazoAnteriorFecha = txtFinEmbarazoAntFecha.Text.Trim();
                paciente.FinEmbarazoAnteriorTerminacion = txtFinEmbarazoAntTerm.Text.Trim();
                paciente.EmbarazoNoDeseado = chkEmbarazoNoDeseado.Checked;
                paciente.UsabaMetodoAnticonceptivo = chkUsabaAnticonceptivo.Checked;
                paciente.MetodoAnticonceptivoEmpleado = txtMetodoAnticonceptivoEmp.Text.Trim();
                paciente.PesoAnteriorKg = pesoAnt;
                paciente.EdadGestacionalTexto = txtEdadGestacional.Text.Trim();

                paciente.FUR = dtpFUR.Value;
                paciente.CalcularFPP();

                // Consumo y riesgos por trimestre
                paciente.Trimestre1 = new ConsumoYRiesgoTrimestre { FumaActivamente = chkT1FumaAct.Checked, FumaPasivamente = chkT1FumaPas.Checked, Drogas = chkT1Drogas.Checked, Alcohol = chkT1Alcohol.Checked, Violencia = chkT1Violencia.Checked };
                paciente.Trimestre2 = new ConsumoYRiesgoTrimestre { FumaActivamente = chkT2FumaAct.Checked, FumaPasivamente = chkT2FumaPas.Checked, Drogas = chkT2Drogas.Checked, Alcohol = chkT2Alcohol.Checked, Violencia = chkT2Violencia.Checked };
                paciente.Trimestre3 = new ConsumoYRiesgoTrimestre { FumaActivamente = chkT3FumaAct.Checked, FumaPasivamente = chkT3FumaPas.Checked, Drogas = chkT3Drogas.Checked, Alcohol = chkT3Alcohol.Checked, Violencia = chkT3Violencia.Checked };

                // Vacunas 
                if (chkVacTetanos.Checked) paciente.Vacunas.Add(new RegistroVacuna { NombreVacuna = "Tétanos/difteria", Aplicada = true, FechaAplicacion = txtFechaTetanos.Text.Trim() });
                if (chkVacTdpa.Checked) paciente.Vacunas.Add(new RegistroVacuna { NombreVacuna = "Tdpa", Aplicada = true, FechaAplicacion = txtFechaTdpa.Text.Trim() });
                if (chkVacInfluenza.Checked) paciente.Vacunas.Add(new RegistroVacuna { NombreVacuna = "Influenza", Aplicada = true, FechaAplicacion = txtFechaInfluenza.Text.Trim() });
                if (chkVacRubeola.Checked) paciente.Vacunas.Add(new RegistroVacuna { NombreVacuna = "Rubéola", Aplicada = true, FechaAplicacion = txtFechaRubeola.Text.Trim() });
                if (chkVacHepB.Checked) paciente.Vacunas.Add(new RegistroVacuna { NombreVacuna = "Hepatitis B", Aplicada = true, FechaAplicacion = txtFechaHepB.Text.Trim() });
                if (chkVacHepA.Checked) paciente.Vacunas.Add(new RegistroVacuna { NombreVacuna = "Hepatitis A", Aplicada = true, FechaAplicacion = txtFechaHepA.Text.Trim() });
                if (chkVacCovid.Checked) paciente.Vacunas.Add(new RegistroVacuna { NombreVacuna = "Covid", Aplicada = true, FechaAplicacion = txtFechaCovid.Text.Trim() });

                // Perfil Prenatal completo
                string vihRes = chkVioVih.Checked ? $"Realizada. <20sem: {txtObsVihMenor.Text.Trim()} | >=20sem: {txtObsVihMayor.Text.Trim()}" : "No se realizó";
                string sifilisRes = chkVioSifilis.Checked ? $"Realizada. <20sem: {txtObsSifMenor.Text.Trim()} | >=20sem: {txtObsSifMayor.Text.Trim()}" : "No se realizó";

                paciente.Prenatal = new PerfilPrenatal
                {
                    CitologiaFechaToma = txtCitoToma.Text.Trim(),
                    CitologiaFechaResultado = txtCitoResFecha.Text.Trim(),
                    CitologiaResultado = txtCitoResultado.Text.Trim(),
                    GrupoSanguineo = txtGrupoSanguineo.Text.Trim(),
                    HemoglobinaMenor20Sem = txtHemoMenor.Text.Trim(),
                    HemoglobinaMayor20Sem = txtHemoMayor.Text.Trim(),
                    Chagas = txtChagas.Text.Trim(),
                    BacteriuriaMenor20Sem = txtBactMenor.Text.Trim(),
                    BacteriuriaMayor20Sem = txtBactMayor.Text.Trim(),
                    GlucemiaAyunoMenor30Sem = txtGlucemia.Text.Trim(),
                    EstreptococosB_30_37Sem = txtEstreptococo.Text.Trim(),
                    VIH = vihRes,
                    Sifilis = sifilisRes,
                    UltrasonografiaObs = txtUltraObs.Text.Trim()
                };

                // Solicitud de laboratorios específicos 
                string fechaLab = string.IsNullOrWhiteSpace(txtFechaLabDefault.Text) ? DateTime.Now.ToShortDateString() : txtFechaLabDefault.Text.Trim();

                if (chkLabHb.Checked) paciente.LaboratoriosEmbarazada.Add(new SolicitudLaboratorioEmbarazada { Fecha = fechaLab, NombreExamen = "HT - HB", TipoMuestra = "sangre" });
                if (chkLabSifilis.Checked) paciente.LaboratoriosEmbarazada.Add(new SolicitudLaboratorioEmbarazada { Fecha = fechaLab, NombreExamen = "Serología de sífilis", TipoMuestra = "sangre" });
                if (chkLabUrocultivo.Checked) paciente.LaboratoriosEmbarazada.Add(new SolicitudLaboratorioEmbarazada { Fecha = fechaLab, NombreExamen = "Urocultivo", TipoMuestra = "orina" });
                if (chkLabVih.Checked) paciente.LaboratoriosEmbarazada.Add(new SolicitudLaboratorioEmbarazada { Fecha = fechaLab, NombreExamen = "VIH", TipoMuestra = "sangre" });
                if (chkLabGlucosa.Checked) paciente.LaboratoriosEmbarazada.Add(new SolicitudLaboratorioEmbarazada { Fecha = fechaLab, NombreExamen = "Glucosa", TipoMuestra = "sangre" });
                if (chkLabOrina.Checked) paciente.LaboratoriosEmbarazada.Add(new SolicitudLaboratorioEmbarazada { Fecha = fechaLab, NombreExamen = "General de orina", TipoMuestra = "orina" });

                // Diagnóstico, tipo de consulta y tratamiento
                paciente.DiagnosticoTexto = txtDiagnostico.Text.Trim();
                paciente.TipoDiagnostico = cmbTipoDiagnostico.SelectedItem.ToString();
                paciente.TipoConsulta = cmbTipoConsulta.SelectedItem.ToString();
                paciente.Recomendaciones = txtRecomendaciones.Text.Trim();

                if (!string.IsNullOrWhiteSpace(txtReceta.Text))
                {
                    paciente.Receta.Add(new Medicamento("Suplemento / Medicamento Prenatal", txtReceta.Text.Trim()));
                }

                // Guardar en base de datos mediante Entity Framework
                repositorio.RegistrarPaciente(paciente);

                MessageBox.Show($"¡Expediente de Paciente Embarazada guardado con éxito!\nFPP Calculada: {paciente.FPP.ToShortDateString()}", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                this.Close();
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ocurrió un error al guardar el registro: {ex.Message}", "Error de Sistema", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }
    }

        private bool ValidarCamposFormulario(out int edad, out double temp, out int fr, out int fc, out int pulso, out double spo2, out double peso, out double talla, out double cir)
        {
            edad = 0; temp = 0; fr = 0; fc = 0; pulso = 0; spo2 = 0; peso = 0; talla = 0; cir = 0;

            try
            {
                if (!int.TryParse(txtEdad.Text.Trim(), out edad))
                {
                    MessageBox.Show("La edad ingresada debe ser un número entero válido.", "Error de Validación", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    txtEdad.Focus();
                    return false;
                }
                ValidadorClinico.ValidarDatosGenerales(
                    txtExpediente.Text.Trim(),
                    txtNombre.Text.Trim(),
                    edad,
                    cmbSexo?.SelectedItem?.ToString() ?? "Femenino"
                );

                ValidadorClinico.ValidarMotivoConsulta(
                    txtConsultaPor.Text.Trim(),
                    txtPresenteEnfermedad.Text.Trim()
                );
                if (!double.TryParse(txtTemp.Text.Trim(), out temp) ||
                    !int.TryParse(txtFR.Text.Trim(), out fr) ||
                    !int.TryParse(txtFC.Text.Trim(), out fc) ||
                    !int.TryParse(txtPulso.Text.Trim(), out pulso) ||
                    !double.TryParse(txtSpO2.Text.Trim(), out spo2) ||
                    !double.TryParse(txtPeso.Text.Trim(), out peso) ||
                    !double.TryParse(txtTalla.Text.Trim(), out talla) ||
                    !double.TryParse(txtCirAbdominal.Text.Trim(), out cir))
                {
                    MessageBox.Show("Uno o más campos de Signos Vitales o Medidas contienen formatos numéricos incorrectos o están vacíos.", "Error de Formato", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return false;
                }

                ValidadorClinico.ValidarSignosVitales(temp, fr, fc, pulso, txtPA.Text.Trim(), spo2, peso, talla, cir);

                return true;
            }
            catch (ExcepcionValidacionClinica ex)
            {
                MessageBox.Show(ex.Message, "Validación Clínica Requerida", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return false;
            }
        }
        private void BGuardar_Click(object sender, EventArgs e)
        {
            //Ejecutar validaciones generales y parseo de signos vitales
            if (!ValidarCamposFormulario(out int edad, out double temp, out int fr, out int fc, out int pulso, out double spo2, out double peso, out double talla, out double cir))
            {
                return;
            }

            //Validación de rango de edad específica para Paciente Infantil
            if (edad < 0 || edad > 11)
            {
                MessageBox.Show("Para el paciente infantil, la edad debe estar comprendida entre 0 y 11 años.", "Error de Rango", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                txtEdad.Focus();
                return;
            }

            try
            {
                //Instanciación de la subclase PacienteInfantil con datos ya validados
                PacienteInfantil paciente = new PacienteInfantil(
                    txtNombre.Text.Trim(),
                    edad,
                    txtExpediente.Text.Trim(),
                    cmbSexo.SelectedItem.ToString(),
                    string.IsNullOrWhiteSpace(txtDireccion.Text) ? "N/A" : txtDireccion.Text.Trim(),
                    string.IsNullOrWhiteSpace(txtResponsable.Text) ? "N/A" : txtResponsable.Text.Trim()
                );

                //Motivo de consulta
                paciente.ConsultaPor = txtConsultaPor.Text.Trim();
                paciente.PresenteEnfermedad = txtPresenteEnfermedad.Text.Trim();

                //Mapeo del Examen Físico completo por regiones
                string[] regiones = new string[]
                {
            "Apariencia general", "Cabeza", "Cuello",
            "Tórax y pulmonar", "Cardíaco", "Abdomen",
            "Columna Vertebral", "Región Glútea", "Genitales",
            "Extremidades", "Piel y anexos", "Neurologico"
                };

                foreach (var reg in regiones)
                {
                    string estadoSeleccionado = cmbsExamenFisicoEstado[reg].SelectedItem.ToString();
                    bool esNormal = (estadoSeleccionado == "Normal");
                    string detalle = txtsExamenFisicoDetalle[reg].Text.Trim();

                    if (string.IsNullOrWhiteSpace(detalle))
                    {
                        detalle = esNormal ? "Sin hallazgos / Normal" : "Alteración no especificada";
                    }

                    paciente.ExamenFisico[reg] = new HallazgoExamen
                    {
                        EsNormal = esNormal,
                        Detalle = detalle
                    };
                }

                //Asignación de signos vitales y medidas ya procesados
                paciente.TemperaturaCelsius = temp;
                paciente.FrecuenciaRespiratoria = fr;
                paciente.FrecuenciaCardiaca = fc;
                paciente.Pulso = pulso;
                paciente.PresionArterial = string.IsNullOrWhiteSpace(txtPA.Text) ? "90/60" : txtPA.Text.Trim();
                paciente.SaturacionOxigeno = spo2;
                paciente.PesoKg = peso;
                paciente.TallaCm = talla;
                paciente.CircunferenciaAbdominalCm = cir;

                //Asignación de campos propios de PacienteInfantil (Micronutrientes y Protección Social)
                paciente.VitaminaA = chkVitaminaA.Checked;
                paciente.Multivitaminas = chkMultivitaminas.Checked;
                paciente.Hierro = chkHierro.Checked;
                paciente.Mebendazol = chkMebendazol.Checked;
                paciente.Albendazol = chkAlbendazol.Checked;

                paciente.EsquemaVacunacionCompleto = chkEsquemaVacunas.Checked;
                paciente.SufreMaltratoOViolencia = chkMaltrato.Checked;
                paciente.CumpleDerechosNino = chkDerechosNino.Checked;
                paciente.RiesgoSocial = chkRiesgoSocial.Checked;

                //Diagnóstico, tipo de consulta y tratamiento
                paciente.DiagnosticoTexto = txtDiagnostico.Text.Trim();
                paciente.TipoDiagnostico = cmbTipoDiagnostico.SelectedItem.ToString();
                paciente.TipoConsulta = cmbTipoConsulta.SelectedItem.ToString();
                paciente.Recomendaciones = txtRecomendaciones.Text.Trim();

                if (!string.IsNullOrWhiteSpace(txtReceta.Text))
                {
                    paciente.Receta.Add(new Medicamento("Medicamento Pediátrico", txtReceta.Text.Trim()));
                }

                //Guardar mediante Repositorio / Entity Framework
                repositorio.RegistrarPaciente(paciente);

                MessageBox.Show("¡Expediente de Paciente Infantil guardado con éxito!", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                this.Close();
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ocurrió un error al guardar el registro: {ex.Message}", "Error de Sistema", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

                double.TryParse(txtTemp.Text, out double temp);
                int.TryParse(txtFR.Text, out int fr);
                int.TryParse(txtFC.Text, out int fc);
                int.TryParse(txtPulso.Text, out int pulso);
                double.TryParse(txtSpO2.Text, out double spo2);
                double.TryParse(txtPeso.Text, out double peso);
                double.TryParse(txtTalla.Text, out double talla);
                double.TryParse(txtCirAbdominal.Text, out double cir);

                paciente.TemperaturaCelsius = temp;
                paciente.FrecuenciaRespiratoria = fr;
                paciente.FrecuenciaCardiaca = fc;
                paciente.Pulso = pulso;
                paciente.PresionArterial = string.IsNullOrWhiteSpace(txtPA.Text) ? "120/80" : txtPA.Text.Trim();
                paciente.SaturacionOxigeno = spo2;
                paciente.PesoKg = peso;
                paciente.TallaCm = talla;
                paciente.CircunferenciaAbdominalCm = cir;

                // Propiedades gineco-obstétricas específicas
                paciente.MetodoAnticonceptivo = cmbMetodoAnticonceptivo.SelectedItem.ToString();
                paciente.FechaUltimaMenstruacion = dtpFUM.Value.ToShortDateString();
                paciente.FechaUltimoParto = dtpFUP.Value.ToShortDateString();
                paciente.FormulaObstetrica = string.IsNullOrWhiteSpace(txtFormulaObstetrica.Text) ? "G0 P0 A0 V0" : txtFormulaObstetrica.Text.Trim();

                // Diagnóstico, tipo de consulta y tratamiento
                paciente.DiagnosticoTexto = txtDiagnostico.Text.Trim();
                paciente.TipoDiagnostico = cmbTipoDiagnostico.SelectedItem.ToString();
                paciente.TipoConsulta = cmbTipoConsulta.SelectedItem.ToString();
                paciente.Recomendaciones = txtRecomendaciones.Text.Trim();

                if (!string.IsNullOrWhiteSpace(txtReceta.Text))
                {
                    paciente.Receta.Add(new Medicamento("Medicamento Ginecológico", txtReceta.Text.Trim()));
                }

                // Guardar en base de datos mediante Entity Framework
                repositorio.RegistrarPaciente(paciente);

                MessageBox.Show("¡Expediente de Mujer en Edad Fértil guardado con éxito!", "Éxito", MessageBoxButtons.OK, MessageBoxIcon.Information);
                this.Close();
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Ocurrió un error al guardar el registro: {ex.Message}", "Error de Sistema", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }
    }

    public class RepositorioSql : IRepositorioPaciente
    {
        private readonly ConexionBD _conexionBD = new ConexionBD();

        public void RegistrarPaciente(Paciente nuevoPaciente)
        {
            using (SqlConnection conexion = _conexionBD.ObtenerConexion())
            {
                conexion.Open();
                SqlTransaction transaccion = conexion.BeginTransaction();

                try
                {
                    // Guardar o actualizar en la tabla Pacientes
                    string queryPaciente = @"
                        IF NOT EXISTS (SELECT 1 FROM Pacientes WHERE NumeroExpediente = @NumeroExpediente)
                        BEGIN
                            INSERT INTO Pacientes (NumeroExpediente, Nombre, Edad, Sexo, Direccion, Responsable, TipoPaciente)
                            VALUES (@NumeroExpediente, @Nombre, @Edad, @Sexo, @Direccion, @Responsable, @TipoPaciente)
                        END
                        ELSE
                        BEGIN
                            UPDATE Pacientes 
                            SET Nombre = @Nombre, Edad = @Edad, Sexo = @Sexo, Direccion = @Direccion, Responsable = @Responsable, TipoPaciente = @TipoPaciente
                            WHERE NumeroExpediente = @NumeroExpediente
                        END";

                    using (SqlCommand cmdPaciente = new SqlCommand(queryPaciente, conexion, transaccion))
                    {
                        cmdPaciente.Parameters.AddWithValue("@NumeroExpediente", nuevoPaciente.NumeroExpediente);
                        cmdPaciente.Parameters.AddWithValue("@Nombre", nuevoPaciente.Nombre);
                        cmdPaciente.Parameters.AddWithValue("@Edad", nuevoPaciente.Edad);
                        cmdPaciente.Parameters.AddWithValue("@Sexo", nuevoPaciente.Sexo ?? (object)DBNull.Value);
                        cmdPaciente.Parameters.AddWithValue("@Direccion", nuevoPaciente.Direccion ?? (object)DBNull.Value);
                        cmdPaciente.Parameters.AddWithValue("@Responsable", nuevoPaciente.Responsable ?? (object)DBNull.Value);
                        cmdPaciente.Parameters.AddWithValue("@TipoPaciente", nuevoPaciente.GetType().Name);

                        cmdPaciente.ExecuteNonQuery();
                    }

                    // Guardar datos de la Consulta
                    string queryConsulta = @"
                        INSERT INTO Consultas (
                            NumeroExpediente, FechaConsulta, ConsultaPor, PresenteEnfermedad,
                            TemperaturaCelsius, FrecuenciaRespiratoria, FrecuenciaCardiaca, Pulso,
                            PresionArterial, SaturacionOxigeno, PesoKg, TallaCm, CircunferenciaAbdominalCm,
                            TipoDiagnostico, DiagnosticoTexto, TipoConsulta, Recomendaciones
                        )
                        OUTPUT INSERTED.IdConsulta
                        VALUES (
                            @NumeroExpediente, @FechaConsulta, @ConsultaPor, @PresenteEnfermedad,
                            @TemperaturaCelsius, @FrecuenciaRespiratoria, @FrecuenciaCardiaca, @Pulso,
                            @PresionArterial, @SaturacionOxigeno, @PesoKg, @TallaCm, @CircunferenciaAbdominalCm,
                            @TipoDiagnostico, @DiagnosticoTexto, @TipoConsulta, @Recomendaciones
                        )";

                    int idConsulta = 0;
                    using (SqlCommand cmdConsulta = new SqlCommand(queryConsulta, conexion, transaccion))
                    {
                        cmdConsulta.Parameters.AddWithValue("@NumeroExpediente", nuevoPaciente.NumeroExpediente);
                        cmdConsulta.Parameters.AddWithValue("@FechaConsulta", nuevoPaciente.FechaConsulta);
                        cmdConsulta.Parameters.AddWithValue("@ConsultaPor", (object)nuevoPaciente.ConsultaPor ?? DBNull.Value);
                        cmdConsulta.Parameters.AddWithValue("@PresenteEnfermedad", (object)nuevoPaciente.PresenteEnfermedad ?? DBNull.Value);
                        cmdConsulta.Parameters.AddWithValue("@TemperaturaCelsius", nuevoPaciente.TemperaturaCelsius);
                        cmdConsulta.Parameters.AddWithValue("@FrecuenciaRespiratoria", nuevoPaciente.FrecuenciaRespiratoria);
                        cmdConsulta.Parameters.AddWithValue("@FrecuenciaCardiaca", nuevoPaciente.FrecuenciaCardiaca);
                        cmdConsulta.Parameters.AddWithValue("@Pulso", nuevoPaciente.Pulso);
                        cmdConsulta.Parameters.AddWithValue("@PresionArterial", (object)nuevoPaciente.PresionArterial ?? DBNull.Value);
                        cmdConsulta.Parameters.AddWithValue("@SaturacionOxigeno", nuevoPaciente.SaturacionOxigeno);
                        cmdConsulta.Parameters.AddWithValue("@PesoKg", nuevoPaciente.PesoKg);
                        cmdConsulta.Parameters.AddWithValue("@TallaCm", nuevoPaciente.TallaCm);
                        cmdConsulta.Parameters.AddWithValue("@CircunferenciaAbdominalCm", nuevoPaciente.CircunferenciaAbdominalCm);
                        cmdConsulta.Parameters.AddWithValue("@TipoDiagnostico", (object)nuevoPaciente.TipoDiagnostico ?? DBNull.Value);
                        cmdConsulta.Parameters.AddWithValue("@DiagnosticoTexto", (object)nuevoPaciente.DiagnosticoTexto ?? DBNull.Value);
                        cmdConsulta.Parameters.AddWithValue("@TipoConsulta", (object)nuevoPaciente.TipoConsulta ?? DBNull.Value);
                        cmdConsulta.Parameters.AddWithValue("@Recomendaciones", (object)nuevoPaciente.Recomendaciones ?? DBNull.Value);

                        idConsulta = (int)cmdConsulta.ExecuteScalar();
                    }

                    // Guardar Examen Físico
                    foreach (var examen in nuevoPaciente.ExamenFisico)
                    {
                        string queryExamen = @"
                            INSERT INTO ExamenesFisicos (IdConsulta, Region, EsNormal, Detalle)
                            VALUES (@IdConsulta, @Region, @EsNormal, @Detalle)";

                        using (SqlCommand cmdExamen = new SqlCommand(queryExamen, conexion, transaccion))

                            cmdExamen.Parameters.AddWithValue("@IdConsulta", idConsulta);
                            cmdExamen.Parameters.AddWithValue("@Region", examen.Key);

                            cmdExamen.ExecuteNonQuery();
                        }
                    }

                    foreach (var med in nuevoPaciente.Receta)

                         string queryReceta = @"
                           INSERT INTO Recetas (IdConsulta, IndicacionesMedicamento)
                           VALUES (@IdConsulta, @IndicacionesMedicamento)";

                        using (SqlCommand cmdReceta = new SqlCommand(queryReceta, conexion, transaccion))

                            cmdReceta.Parameters.AddWithValue("@IdConsulta", idConsulta);
                            cmdReceta.Parameters.AddWithValue("@IndicacionesMedicamento", (object)med.CantidadYDosis ?? DBNull.Value);

                            cmdAM.Parameters.AddWithValue("@CategoriaDependencia", (object)adultoMayor.CategoriaDependencia ?? DBNull.Value);

                            cmdAM.ExecuteNonQuery();
                        }
                    }
                    else if (nuevoPaciente is PacienteEmbarazada embarazada)
                    {
                          string queryEmbarazada = @"
                                 INSERT INTO DetalleEmbarazada (
                                 IdConsulta, 
                                 FormulaObstetrica, 
                                 FinEmbarazoAntFecha, 
                                 FinEmbarazoAntTerminacion, 
                                 EmbarazoNoDeseado,

                                 GrupoSanguineo,

                                 UltrasonografiaObs,
                                 -- Vacunas y fechas
                                 FechaTetanos, FechaTdpa, FechaInfluenza, FechaRubeola, FechaHepB, FechaHepA, FechaCovid,
                                 -- Perfil Prenatal y Exámenes
                                 CitologiaFechaToma, CitologiaFechaResultado, CitologiaResultado,
                                 HemoglobinaMenor20Sem, HemoglobinaMayor20Sem,

                                              @UsabaMetodoAnticonceptivo,

                            cmdEmb.Parameters.AddWithValue("@EmbarazoNoDeseado", embarazada.EmbarazoNoDeseado);

        public Paciente BuscarPorExpediente(string numExpediente)
        {
            using (SqlConnection conexion = _conexionBD.ObtenerConexion())
            {
                string query = "SELECT NumeroExpediente, Nombre, Edad, Sexo, Direccion, Responsable, TipoPaciente FROM Pacientes WHERE NumeroExpediente = @Expediente";
                SqlCommand cmd = new SqlCommand(query, conexion);
                cmd.Parameters.AddWithValue("@Expediente", numExpediente);

                conexion.Open();
                using (SqlDataReader reader = cmd.ExecuteReader())
                {
                    if (reader.Read())
                    {
                        string tipo = reader["TipoPaciente"].ToString();
                        string nombre = reader["Nombre"].ToString();
                        int edad = Convert.ToInt32(reader["Edad"]);
                        string exp = reader["NumeroExpediente"].ToString();
                        string sexo = reader["Sexo"].ToString();
                        string dir = reader["Direccion"].ToString();
                        string resp = reader["Responsable"].ToString();

                        switch (tipo)
                        {
                            case nameof(PacienteInfantil):
                                return new PacienteInfantil(nombre, edad, exp, sexo, dir, resp);
                            case nameof(PacienteAdolescente):
                                return new PacienteAdolescente(nombre, edad, exp, sexo, dir, resp);
                            case nameof(PacienteAdultoMayor):
                                return new PacienteAdultoMayor(nombre, edad, exp, sexo, dir, resp);
                            case nameof(PacienteEmbarazada):
                                return new PacienteEmbarazada(nombre, edad, exp, dir, resp);
                            case nameof(PacienteMujerEdadFertil):
                                return new PacienteMujerEdadFertil(nombre, edad, exp, dir, resp);
                            default:
                                return new PacienteCronico(nombre, edad, exp, sexo, dir, resp);
                        }
                    }
                }
            }
            return null;
        }

        public static SesionUsuario Instancia
        {
            get
            {
                lock (_lock)
                {
                    if (_instancia == null)
                    {
                        _instancia = new SesionUsuario();
                    }
                    return _instancia;
                }
            }
        }

        {
            NombreUsuario = usuario;
            Rol = rol;
            Console.WriteLine($"\n[SESIÓN INICIADA]: Usuario '{NombreUsuario}' con rol [{Rol}].");
        }

        public void CerrarSesion()
        {
            Console.WriteLine($"\n[SESIÓN CERRADA]: Salió el usuario '{NombreUsuario}'.");
            NombreUsuario = null;
            Rol = null;
        }

    public static class ValidadorClinico
    {
        //Validar Datos Generales Básicos
        public static void ValidarDatosGenerales(string expediente, string nombre, int edad, string sexo)
        {
            if (string.IsNullOrWhiteSpace(expediente))
            {
                throw new ExcepcionValidacionClinica("El 'Número de Expediente' es obligatorio.");
            }

            if (string.IsNullOrWhiteSpace(nombre))
            {
                throw new ExcepcionValidacionClinica("El 'Nombre del Paciente' es obligatorio.");
            }

            if (string.IsNullOrWhiteSpace(sexo))
            {
                throw new ExcepcionValidacionClinica("Debe seleccionar el 'Sexo' o 'Género' del paciente.");
            }
        }

        //Validar Motivo de Consulta
        public static void ValidarMotivoConsulta(string consultaPor, string presenteEnfermedad)
        {
            if (string.IsNullOrWhiteSpace(consultaPor))
            {
                throw new ExcepcionValidacionClinica("El campo 'Consulta Por' es obligatorio.");
            }

            if (string.IsNullOrWhiteSpace(presenteEnfermedad))
            {
                throw new ExcepcionValidacionClinica("El campo 'Presente Enfermedad' es obligatorio.");
            }
        }

        //Validar Signos Vitales y Medidas Antropométricas
        public static void ValidarSignosVitales(
            double temp, int fr, int fc, int pulso, string pa, double spo2, double peso, double talla, double cirAbdomen)
        {
            if (temp < 30.0 || temp > 45.0)
            {
                throw new ExcepcionValidacionClinica($"La temperatura ({temp}°C) no está en un rango biológico válido (30°C - 45°C).");
            }

            if (fr < 5 || fr > 100)
            {
                throw new ExcepcionValidacionClinica($"La frecuencia respiratoria ({fr} rpm) no es válida (Rango: 5 - 100 rpm).");
            }

            if (fc < 30 || fc > 250)
            {
                throw new ExcepcionValidacionClinica($"La frecuencia cardíaca ({fc} lpm) no es válida (Rango: 30 - 250 lpm).");
            }

            if (pulso < 30 || pulso > 250)
            {
                throw new ExcepcionValidacionClinica($"El pulso ({pulso} ppm) no es válido (Rango: 30 - 250 ppm).");
            }

            if (string.IsNullOrWhiteSpace(pa) || !pa.Contains("/"))
            {
                throw new ExcepcionValidacionClinica("La Presión Arterial debe tener un formato válido (ej. 120/80).");
            }

            if (spo2 < 50.0 || spo2 > 100.0)
            {
                throw new ExcepcionValidacionClinica($"La saturación de oxígeno ({spo2}%) no es válida (Rango: 50% - 100%).");
            }

            if (peso <= 0 || peso > 400.0)
            {
                throw new ExcepcionValidacionClinica($"El peso ({peso} kg) no es válido.");
            }

            if (talla <= 0 || talla > 250.0)
            {
                throw new ExcepcionValidacionClinica($"La talla ({talla} cm) no es válida.");
            }

            if (cirAbdomen < 0 || cirAbdomen > 300.0)
            {
                throw new ExcepcionValidacionClinica("La circunferencia abdominal ingresada no es válida.");
            }
        }
    }
